  AND r2.rating >= 4
  AND common IN likedMovies

WITH target, other, count(DISTINCT common) AS similarity
WHERE similarity > 0

MATCH (other)-[r3:RATED]->(candidate:Movie)
WHERE r3.rating >= 4
  AND NOT EXISTS {
    MATCH (target)-[:RATED]->(candidate)
  }

     sum(similarity * r3.rating) / sum(similarity) AS score,
     count(DISTINCT other) AS recommendedBy

    candidate.movieId AS movieId,
    candidate.title AS recommendedMovie,
    round(score * 100) / 100.0 AS score,
    recommendedBy
